name: Refresh animated profile

on:
  schedule:
    - cron: "17 */6 * * *"
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: animated-profile-refresh
  cancel-in-progress: false

jobs:
  refresh:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - name: Checkout profile repository
        uses: actions/checkout@v4
      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: pip
      - name: Cache portrait model
        uses: actions/cache@v4
        with:
          path: ~/.u2net
          key: rembg-u2net-v2
      - name: Install renderer
        run: python -m pip install --upgrade pip && python -m pip install -r requirements.txt
      - name: Generate complete profile
        env:
          GH_LOGIN: ${{ github.repository_owner }}
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: python scripts/generate_profile.py
      - name: Commit only changed output
        run: |
          if [ -z "$(git status --porcelain)" ]; then
            echo "Profile already current."
            exit 0
          fi
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add README.md assets/
          git commit -m "profile: refresh animated interface"
          git push
